CHATGPT Ocean Scene


let bubbles = [];
let sparks = [];
let jellies = [];

function setup() {
  createCanvas(800, 600);
  angleMode(RADIANS);

  for (let i = 0; i < 90; i++) {
    bubbles.push({
      x: random(width),
      y: random(height),
      r: random(4, 12),
      v: random(0.6, 2),
      wob: random(TWO_PI)
    });
  }

  for (let i = 0; i < 220; i++) {
    sparks.push({
      theta: random(TWO_PI),
      r: random(15, 120),
      w: random(-0.02, 0.02),
      sz: random(3, 6)
    });
  }

  for (let i = 0; i < 5; i++) {
    jellies.push({
      x: random(150, 650),
      y: random(200, 500),
      phase: random(TWO_PI),
      s: random(0.9, 1.3)
    });
  }
}

function draw() {
  drawBrightOcean();
  drawRocks();
  drawPortal(550, 300, 220);
  drawBubbles();
  drawJellies();
}

function drawBrightOcean() {
  noStroke();
  for (let y = 0; y < height; y++) {
    let t = y / height;
    fill(
      lerp(40, 120, t),
      lerp(180, 230, t),
      lerp(220, 255, t)
    );
    rect(0, y, width, 1);
  }
}

function drawRocks() {
  noStroke();
  fill(40, 70, 90);
  beginShape();
  vertex(0, height);
  vertex(0, 460);
  vertex(120, 430);
  vertex(260, 470);
  vertex(400, 440);
  vertex(600, 480);
  vertex(760, 450);
  vertex(width, 480);
  vertex(width, height);
  endShape(CLOSE);
}

function drawPortal(cx, cy, size) {
  noStroke();

  for (let i = 5; i >= 1; i--) {
    fill(200, 245, 255, 30);
    ellipse(cx, cy, size + i * 50);
  }

  for (let r = size; r > 40; r -= 18) {
    fill(120, 220, 255, 120);
    ellipse(cx, cy, r);
  }

  fill(255);
  ellipse(cx, cy, 60);

  for (let s of sparks) {
    s.theta += s.w;
    let x = cx + cos(s.theta) * s.r;
    let y = cy + sin(s.theta) * s.r;
    fill(255, 240, 180);
    circle(x, y, s.sz);
  }
}

function drawBubbles() {
  noStroke();
  for (let b of bubbles) {
    b.y -= b.v;
    b.x += sin(frameCount * 0.02 + b.wob);

    if (b.y < -20) b.y = height + 20;

    fill(255, 255, 255, 180);
    circle(b.x, b.y, b.r);
  }
}

function drawJellies() {
  for (let j of jellies) {
    j.phase += 0.02;
    let bob = sin(j.phase) * 10;

    push();
    translate(j.x, j.y + bob);
    scale(j.s);

    // glow
    for (let i = 3; i >= 1; i--) {
      fill(255, 200, 255, 30);
      ellipse(0, 0, 120 + i * 20, 80 + i * 15);
    }

    // dome
    fill(255, 220, 255, 160);
    arc(0, 0, 130, 90, PI, TWO_PI);

    // tentacles
    stroke(255, 200, 255, 160);
    strokeWeight(3);
    noFill();
    for (let t = -4; t <= 4; t++) {
      beginShape();
      for (let p = 0; p < 10; p++) {
        curveVertex(
          t * 12 + sin(j.phase + p * 0.6) * 10,
          20 + p * 14
        );
      }
      endShape();
    }
    noStroke();

    pop();
  }
}
